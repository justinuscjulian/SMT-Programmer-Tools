Sub MACROPLAN1_Clean()
    Dim ws As Worksheet
    Dim colG As Range
    Dim colT As Range
    Dim targetCell As Range

    ' Gunakan worksheet aktif
    Set ws = ActiveSheet

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual

    ' 1?? Ubah semua formula di sheet jadi value
    With ws.UsedRange
        .Value = .Value
        .Font.Name = "Calibri"
        .Font.Strikethrough = False
        .Font.Superscript = False
        .Font.Subscript = False
        .Font.Underline = xlUnderlineStyleNone
        .Font.TintAndShade = 0
        .Font.ThemeFont = xlThemeFontMinor
    End With

    ' 2?? Dinamis cari kolom G (kolom ke-7) dan pastikan formatting diterapkan walau kolom bergeser
    On Error Resume Next
    Set colG = ws.Columns(7) ' G = kolom ke-7
    On Error GoTo 0

    If Not colG Is Nothing Then
        colG.Font.Size = 11
        colG.EntireColumn.AutoFit
    End If

    ' 3?? Dinamis cari kolom T (kolom ke-20)
    On Error Resume Next
    Set colT = ws.Columns(20) ' T = kolom ke-20
    On Error GoTo 0

    If Not colT Is Nothing Then
        colT.ColumnWidth = 26.86
    End If

    ' 4?? Zoom tampilan sheet
    ws.Parent.Windows(1).Zoom = 110

    ' 5?? Fokuskan kursor ke sel Y3 secara aman (kolom 25)
    On Error Resume Next
    Set targetCell = ws.Cells(3, 25) ' Baris 3, Kolom Y (25)
    On Error GoTo 0

    If Not targetCell Is Nothing Then
        targetCell.Select
    End If

    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
End Sub
